function switchPage(pageId) {
    document.querySelectorAll('.page-view').forEach(p => p.classList.remove('active'));
    document.getElementById('page-' + pageId).classList.add('active');

    document.querySelectorAll('.links a').forEach(a => a.classList.remove('active'));
    const activeNav = document.getElementById('nav-' + pageId);
    if(activeNav) activeNav.classList.add('active');

    window.scrollTo(0, 0);
}

let interested = 0;

function openModal(){ document.getElementById('modal').classList.add('open'); }
function closeModal(){ document.getElementById('modal').classList.remove('open'); }

const targetDate = new Date(2026, 5, 20, 18, 0, 0).getTime();

function updateCountdown() {
    const now = new Date().getTime();
    const distance = targetDate - now;
    const countdownEl = document.getElementById('countdown');

    if (distance < 0) {
        countdownEl.textContent = "Session Active!";
        return;
    }

    const days = Math.floor(distance / (1000 * 60 * 60 * 24));
    const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
    const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
    const seconds = Math.floor((distance % (1000 * 60)) / 1000);

    let text = "";
    if (days > 0) text += days + "d ";
    text += hours + "h " + minutes + "m " + seconds + "s";
    
    countdownEl.textContent = text;
}
setInterval(updateCountdown, 1000);
updateCountdown();

function saveInterest() {
    const name = document.getElementById('playerName').value.trim();
    const mode = document.getElementById('playerMode').value;
    const msg = document.getElementById('formMsg');

    if (!name) {
        msg.textContent = 'Please enter your in-game name.';
        return;
    }

    const webhookURL = 'https://discord.com/api/webhooks/1560002085896986694/kM6t9VNo9skxMybdhZBRCBe3YASRVTum-2okINlMLXG6a6dnGh1Q8iNoRzmLa6_E2n5l';

    msg.textContent = 'Sending...';

    const payload = {
        content: `New Session Registration! 🎮`,
        embeds: [{
            title: "New Registration",
            color: 7941631,
            fields: [
                { name: "Player / Epic", value: name, inline: true },
                { name: "Selected Mode", value: mode, inline: true }
            ],
            timestamp: new Date().toISOString()
        }]
    };

    fetch(webhookURL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
    })
    .then(response => {
        if (response.ok) {
            interested++;
            document.getElementById('playerCount').textContent = interested + ' Player' + (interested === 1 ? '' : 's') + ' / 100';
            msg.textContent = 'Thank you, ' + name + '! Your registration was sent to Discord.';
            document.getElementById('playerName').value = '';
        } else {
            msg.textContent = 'Error sending to Discord.';
        }
    })
    .catch(error => {
        msg.textContent = 'Network error while sending.';
        console.error(error);
    });
}

function setMode(btn, mode) {
    document.querySelectorAll('.tabs button').forEach(b => b.classList.remove('selected'));
    btn.classList.add('selected');
    document.getElementById('ranks').innerHTML = '<div class="empty">The ' + mode + ' leaderboard will appear here once verified results are submitted.</div>';
}

document.getElementById('modal').addEventListener('click', e => {
    if (e.target.id === 'modal') closeModal();
});
